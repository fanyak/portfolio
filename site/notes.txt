clim <- national_1961_1990 |>
  group_by(yd) |>
  summarise(
    tmed_norm = median(tmed, na.rm = TRUE),
    p5  = quantile(tmed, .05, na.rm = TRUE),
    p95 = quantile(tmed, .95, na.rm = TRUE)
  )

fit_harmonic <- function(y, yd, K, period = 366) {
  fit <- lm(y ~ ., data = bind_cols(y = y, make_harmonics(yd, K, period)))
  predict(fit)
}

clim <- clim |>
  mutate(
    smooth          = fit_harmonic(tmed_norm, yd, best_K),
